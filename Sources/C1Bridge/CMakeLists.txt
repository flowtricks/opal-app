cmake_minimum_required(VERSION 3.20)
project(C1Bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0")
set(CMAKE_OSX_ARCHITECTURES arm64)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SDK_ROOT "${PROJECT_ROOT}/vendor/depthai-install")
find_package(depthai CONFIG REQUIRED PATHS "${SDK_ROOT}/lib/cmake/depthai" NO_DEFAULT_PATH)
find_library(C1_USB_LIBRARY usb-1.0 PATHS "${SDK_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
add_library(C1Bridge SHARED C1Bridge.cpp)
target_include_directories(C1Bridge PUBLIC include PRIVATE "${SDK_ROOT}/lib/cmake/depthai/dependencies/include")
target_link_libraries(C1Bridge PRIVATE depthai::core "${C1_USB_LIBRARY}" "-framework IOKit" "-framework CoreFoundation")
set_target_properties(C1Bridge PROPERTIES INSTALL_RPATH "@loader_path" BUILD_WITH_INSTALL_RPATH TRUE)
enable_testing()
add_executable(C1ControlTests ControlTests.cpp)
target_include_directories(C1ControlTests PRIVATE include "${SDK_ROOT}/lib/cmake/depthai/dependencies/include")
target_link_libraries(C1ControlTests PRIVATE depthai::core "${C1_USB_LIBRARY}" "-framework IOKit" "-framework CoreFoundation")
set_target_properties(C1ControlTests PROPERTIES BUILD_RPATH "${SDK_ROOT}/lib")
add_test(NAME C1ControlTests COMMAND C1ControlTests)
